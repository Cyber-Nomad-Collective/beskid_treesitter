i32 sum_to(i32 n) {
    i32 mut acc = 0;
    for i in range(0, n) {
        acc = acc + i;
    }
    return acc;
}

i32 max(i32 a, i32 b) {
    if a > b {
        return a;
    }
    return b;
}
